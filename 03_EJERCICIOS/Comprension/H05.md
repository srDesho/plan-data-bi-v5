**1. Recuperar un vendedor.**

```python
# filas completas observadas: (ruta, vendedor)
completas = [("R01", "V1"), ("R01", "V1"), ("R02", "V2"), ("R02", "V3"), ("R03", "V4")]
# filas con el vendedor vacío (ruta): se recupera SOLO si la ruta tiene exactamente un vendedor en las filas completas
vacias = ["R01", "R02", "R03", "R05"]
mapa = {}
for ruta, v in completas:
    mapa.setdefault(ruta, set()).add(v)
for ruta in vacias:
    candidatos = mapa.get(ruta, set())
    print(ruta, sorted(candidatos)[0] if len(candidatos) == 1 else "sin regla")
```
